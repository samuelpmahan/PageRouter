function byId(id){
  const visit=value=>{if(!value||typeof value!=='object')return null;if(value.attrs?.id===id)return value;for(const child of value.children||[]){const found=visit(child);if(found)return found;}return null;};

}

function verifyBio(stage,attempted,expected){
  const before=calls.length;
  assert.equal(controller.updateDraft('bio','x'.repeat(attempted)),true,`${stage}: updateDraft accepted`);
  assert.equal(calls.length,before,`${stage}: updateDraft delivered no provider port`);
  const field=byId('hh-profile-bio'),help=byId('hh-profile-bio-help');
  assert.equal(field?.attrs?.maxlength,'500',`${stage}: maxlength exposed`);
  assert.equal(field?.attrs?.['aria-describedby'],'hh-profile-bio-help',`${stage}: help reference bound`);
  assert.ok(help,`${stage}: help node exists`);
  assert.equal(Object.hasOwn(help.attrs,'hidden'),false,`${stage}: help is visible`);
  assert.match(text(help),/up to 500 characters/i);
  assert.equal(controller.state.journey.profile[stage==='create'?'createDraft':'editDraft'].bio.length,expected);
  assert.equal(field.attrs.value.length,expected,`${stage}: clipped/preserved draft remains visible`);
  return {stage,attempted,rendered:field.attrs.value.length,helpText:text(help),providerCallsDuringUpdate:0};
}
