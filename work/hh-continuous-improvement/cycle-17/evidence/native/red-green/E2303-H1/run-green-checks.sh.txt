#!/usr/bin/env bash
set -u
set -o pipefail
set -C

ROOT=/workspace/scratch/8528bb251845/hh-cycle17/H1/E2303/captures

run_check() {
  local capture_dir="$1"
  shift
  date -u +%Y-%m-%dT%H:%M:%SZ > "$capture_dir/start.utc" || return 125
  local rc
  if "$@" > "$capture_dir/stdout.raw" 2> "$capture_dir/stderr.raw"; then
    rc=0
  else
    rc=$?
  fi
  printf '%s\n' "$rc" > "$capture_dir/exit.code" || return 125
  date -u +%Y-%m-%dT%H:%M:%SZ > "$capture_dir/end.utc" || return 125
  return "$rc"
}

run_check "$ROOT/green-01-focused" \
  node --test \
  '--test-name-pattern=public mobile navigation toggle keeps a stable descriptive label across menu and session states' \
  checkpoint/vendor/hh/src/public-site.test.mjs || exit $?

run_check "$ROOT/green-02-public-site" \
  node --test checkpoint/vendor/hh/src/public-site.test.mjs || exit $?

run_check "$ROOT/green-03-controller" \
  node --test checkpoint/vendor/hh/src/public-controller.test.mjs || exit $?
