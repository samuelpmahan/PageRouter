function walk(dir, rel = '') {
  const entries = fs.readdirSync(dir, {withFileTypes: true}).sort((a, b) => a.name.localeCompare(b.name));
  const files = [];
  const dirs = [];
  const special = [];
  for (const entry of entries) {
    const childRel = rel ? `${rel}/${entry.name}` : entry.name;
    const absolute = path.join(dir, entry.name);
    if (entry.isSymbolicLink()) special.push({path: childRel, type: 'symlink'});
    else if (entry.isDirectory()) { dirs.push(childRel); const nested = walk(absolute, childRel); files.push(...nested.files); dirs.push(...nested.dirs); special.push(...nested.special); }
    else if (entry.isFile()) files.push(childRel);
    else special.push({path: childRel, type: 'non-regular'});
  }
  return {files, dirs, special};
}
