const hash = value => createHash('sha256').update(JSON.stringify(value)).digest('hex');
const walk = tree => tree && typeof tree === 'object' ? [tree, ...(tree.children || []).flatMap(walk)] : [];
const byId = (tree, id) => walk(tree).find(node => node.attrs?.id === id) || null;
const text = tree => walk(tree).flatMap(node => node.children || []).filter(value => typeof value === 'string').join(' ');
function makeStorage() {
  const values = new Map();
  return {values, getItem(key) { return values.get(key) ?? null; }, setItem(key, value) { values.set(key, String(value)); }, removeItem(key) { values.delete(key); }};
}
function makeEnv(storage) {
  const base = createFixtureProvider({fixture: true, storage});
  const counts = Object.fromEntries(HH_PORTS.map(port => [port, 0]));
  const provider = {...base, services: Object.freeze(Object.fromEntries(Object.entries(base.services).map(([port, invoke]) => [port, async input => {
    counts[port] += 1;
    return invoke(input);
  }]))) };
  const runtime = createServiceRuntime(provider);
  const community = createCommunity({storage});
  let tree = null;
  const controller = createPublicController({runtime, storage, community, onChange(state, route) {
    tree = renderPublicSite({route, state, journey: state.journey});
  }});
  return {base, counts, provider, runtime, community, controller, get tree() { return tree; }};
}
const storageProjection = storage => [...storage.values.entries()].sort(([a], [b]) => a.localeCompare(b));
const runtimeParts = runtime => runtime.pxc.entries().map(([address, part]) => ({address, kind: typeof part.value === 'function' ? 'function' : address === 'hh.provider' ? 'provider' : 'value'}));
const runtimeState = runtime => runtime.snapshot();
const delta = (before, after) => Object.fromEntries(HH_PORTS.map(port => [port, after[port] - before[port]]));
const hasNonzero = values => Object.values(values).some(value => value !== 0);
const renderedProfile = (tree, ids = ['hh-profile-edit-form', 'hh-profile-name', 'hh-profile-bio', 'hh-profile-public-view']) =>
  Object.fromEntries(ids.map(id => { const node = byId(tree, id); return [id, node ? {tag: node.tag, attrs: node.attrs, text: text(node)} : null]; }));
