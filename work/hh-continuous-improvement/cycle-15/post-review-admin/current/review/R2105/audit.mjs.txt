async function walk(dir, prefix = '') {
  const out = [];
  for (const entry of fs.readdirSync(dir, {withFileTypes: true}).sort((a, b) => a.name.localeCompare(b.name))) {
    const rel = prefix ? `${prefix}/${entry.name}` : entry.name;
    const full = path.join(dir, entry.name);
    const st = fs.lstatSync(full);
    if (st.isSymbolicLink()) out.push({path: rel, type: 'symlink', mode: modeOf(st)});
    else if (st.isDirectory()) out.push(...await walk(full, rel));
    else if (st.isFile()) out.push({path: rel, type: 'file', bytes: st.size, mode: modeOf(st), sha256: sha(fs.readFileSync(full))});
    else out.push({path: rel, type: 'other', mode: modeOf(st)});
  }
  return out;
}
