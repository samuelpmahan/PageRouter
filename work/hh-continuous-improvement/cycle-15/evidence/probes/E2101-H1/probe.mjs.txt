import {createHash} from 'node:crypto';
import {createFixtureProvider, createServiceRuntime, HH_PORTS} from '../../../PageRouter/checkpoint/vendor/hh/services/index.mjs';
import {createCommunity} from '../../../PageRouter/checkpoint/vendor/hh/src/community.mjs';
import {createPublicController} from '../../../PageRouter/checkpoint/vendor/hh/src/public-controller.mjs';
import {renderPublicSite} from '../../../PageRouter/checkpoint/vendor/hh/src/public-site.mjs';

const syntheticEmail = 'hh15-probe@example.test';
const malformedEmail = 'not-an-email';
const digest = value => createHash('sha256').update(typeof value === 'string' ? value : JSON.stringify(value)).digest('hex');
const counts = Object.fromEntries(HH_PORTS.map(port => [port, 0]));
let tree = null;
const provider = createFixtureProvider({fixture: true});
const countedProvider = {
  ...provider,
  services: Object.freeze(Object.fromEntries(Object.entries(provider.services).map(([port, invoke]) => [port, async input => {
    counts[port] = (counts[port] || 0) + 1;
    return invoke(input);
  }]))),
};
const runtime = createServiceRuntime(countedProvider);
const community = createCommunity();
const controller = createPublicController({runtime, community,
  onChange(state, route) { tree = renderPublicSite({route, state, journey: state.journey}); },
});

function nodes(root) {
  if (!root || typeof root !== 'object') return [];
  return [root, ...(root.children || []).flatMap(nodes)];
}
function nodeById(root, id) { return nodes(root).find(node => node.attrs?.id === id) || null; }
function text(root) {
  if (root == null) return '';
  if (typeof root === 'string' || typeof root === 'number') return String(root);
  if (typeof root !== 'object') return '';
  return (root.children || []).map(text).filter(Boolean).join(' ');
}
function inputProjection(root) {
  return nodes(root).filter(node => node.tag === 'input' || node.tag === 'textarea' || node.tag === 'select')
    .map(node => ({tag: node.tag, id: node.attrs?.id ?? null, name: node.attrs?.name ?? null,
      type: node.attrs?.type ?? null, value: node.attrs?.value ?? '', required: node.attrs?.required ?? false,
      autocomplete: node.attrs?.autocomplete ?? null}));
}
function currentForgotView() {
  const main = nodeById(tree, 'hh-main');
  const form = nodeById(tree, 'hh-forgot-form');
  return {page: tree?.attrs?.['data-public-page'] ?? null, mainText: text(main),
    form: form ? {id: form.attrs?.id, submit: form.attrs?.['data-submit'], controls: inputProjection(form),
      buttons: nodes(form).filter(node => node.tag === 'button').map(node => ({type: node.attrs?.type ?? null,
        id: node.attrs?.id ?? null, text: text(node)}))} : null};
}
function partProjection() {
  return runtime.pxc.entries().map(([address, part]) => ({address,
    kind: typeof part.value === 'function' ? 'functionPart' : address === 'hh.provider' ? 'providerPart' : 'valuePart'}));
}
function storageProjection() {
  const snap = runtime.snapshot();
  return {injectedStorage: false, browserSessionStorageAvailable: typeof globalThis.sessionStorage !== 'undefined',
    fixturePersistence: snap.diagnostics?.persistence ?? null,
    communityPersistence: community.snapshot().storageStatus,
    storageKeys: []};
}
function stateProjection() {
  const state = controller.state;
  return {route: state.route, request: state.request, journeyKeys: Object.keys(state.journey).sort(),
    forgotDraft: state.journey.forgotDraft ?? null, outcome: state.journey.outcome,
    loginDraft: state.journey.loginDraft, registrationIdentity: state.journey.registration?.identity,
    contact: state.journey.contact};
}
function runtimeProjection() {
  const snapshot = runtime.snapshot();
  const inspected = runtime.inspect();
  return {snapshotDigest: digest(snapshot), mode: snapshot.mode, fixture: snapshot.fixture,
    persistence: snapshot.diagnostics?.persistence ?? null, sessionStatus: snapshot.session.status,
    registrationStatus: snapshot.registration.status,
    parts: partProjection(), receiptCount: inspected.receipts.length,
    receipts: inspected.receipts.map((receipt, index) => ({index: index + 1, status: receipt.status,
      into: receipt.into, inputNames: receipt.inputNames}))};
}
function countSnapshot() { return {...counts}; }
function delta(before, after) {
  return Object.fromEntries(HH_PORTS.map(port => [port, after[port] - before[port]]));
}
function hasAnyDelta(value) { return Object.values(value).some(number => number !== 0); }
