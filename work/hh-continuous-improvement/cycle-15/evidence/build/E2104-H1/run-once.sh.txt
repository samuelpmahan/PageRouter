#!/usr/bin/env bash
set -u
capture_dir=/workspace/scratch/8528bb251845/hh-cycle15/H1/E2104/captures
for name in stdout.raw stderr.raw exit.code signal.txt start.utc end.utc; do
  if [ -e "$capture_dir/$name" ]; then
    printf 'Refusing to overwrite existing capture: %s\n' "$capture_dir/$name" >&2
    exit 73
  fi
done
cd /workspace/scratch/8528bb251845/PageRouter/checkpoint
date -u +%Y-%m-%dT%H:%M:%SZ > "$capture_dir/start.utc"
/usr/bin/env BAZEL=/workspace/scratch/8528bb251845/tools/bazel bash ci/bazel/run.sh > "$capture_dir/stdout.raw" 2> "$capture_dir/stderr.raw"
build_status=$?
printf '%s\n' "$build_status" > "$capture_dir/exit.code"
if [ "$build_status" -gt 128 ]; then
  signal_number=$((build_status - 128))
  signal_name=$(kill -l "$signal_number" 2>/dev/null || printf 'unknown')
  printf 'possible-signal-%s:%s\n' "$signal_number" "$signal_name" > "$capture_dir/signal.txt"
else
  printf 'none\n' > "$capture_dir/signal.txt"
fi
date -u +%Y-%m-%dT%H:%M:%SZ > "$capture_dir/end.utc"
exit "$build_status"
