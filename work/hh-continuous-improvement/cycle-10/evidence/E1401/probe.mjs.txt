const nodeById = (tree, id) => {
  if (!tree || typeof tree !== 'object') return null;
  if (tree.attrs?.id === id) return tree;
  for (const child of tree.children || []) {
    const match = nodeById(child, id);
    if (match) return match;
  }
  return null;
};
const selectedOptionValue = (tree, id) => nodeById(tree, id)?.children?.find(option => option?.attrs?.selected)?.attrs?.value || '';
const treeText = (tree) => {
  if (typeof tree === 'string') return tree;
  if (!tree || typeof tree !== 'object') return '';
  return (tree.text || '') + (tree.children || []).map(treeText).join(' ');
};
const ephemeralCredential = () => randomBytes(24).toString('base64url');

function makeEnv({failCall = null, pauseFirst = false} = {}) {
  const base = createFixtureProvider({fixture: true});
  let announceStart;
  let release;
  const started = new Promise(resolve => { announceStart = resolve; });
  const gate = new Promise(resolve => { release = resolve; });
  let registrationCalls = 0;
  let loginAliasIndex = null;
  const registrationAliasIndices = [];
  const services = Object.fromEntries(Object.entries(base.services).map(([port, invoke]) => [port, async input => {
    if (port === 'Registration.registerTeacher') {
      registrationCalls += 1;
      const match = /^public-teacher-(\d+)@fixture\.test$/.exec(input.teacher.email);
      registrationAliasIndices.push(match ? Number(match[1]) : null);
      if (pauseFirst && registrationCalls === 1) {
        announceStart();
        await gate;
      }
      if (registrationCalls === failCall?.call) return invoke({...input, failWith: failCall.kind});
      return invoke(input);
    }
    if (port === 'Session.login') {
      const match = /^public-teacher-(\d+)@fixture\.test$/.exec(input.email);
      loginAliasIndex = match ? Number(match[1]) : null;
    }
    return invoke(input);
  }]));
  const provider = {...base, services: Object.freeze(services)};
  const runtime = createServiceRuntime(provider);
  let tree = null;
  const controller = createPublicController({runtime, community: createCommunity(), onChange(state, route) {
    tree = renderPublicSite({route, state, journey: state.journey});
  }});
  return {provider, runtime, controller, get tree() { return tree; }, started, release,
    get registrationCalls() { return registrationCalls; }, get registrationAliasIndices() { return [...registrationAliasIndices]; },
    get loginAliasIndex() { return loginAliasIndex; }};
}

async function prepare(env, suffix) {
  await env.controller.initialize('#/register');
  for (const [key, value] of Object.entries({state: 'Demo Washington', county: 'Demo King', district: 'Demo District', school: 'Demo Academy'})) {
    await env.controller.selectSchool(key, value);
  }
  await env.controller.dispatch('public:terms-consent', {accepted: true});
  const identity = {name: `E1401 Teacher ${suffix}`, email: `e1401-${suffix}@example.test`, phone: '555-0101'};
  for (const [key, value] of Object.entries(identity)) assert.equal(env.controller.updateDraft(key, value), true);
  return identity;
}

function draftRendered(env, identity) {
  const tree = env.tree;
  return {
    form: nodeById(tree, 'hh-register-form') !== null,
    name: nodeById(tree, 'hh-register-name')?.attrs?.value === identity.name,
    email: nodeById(tree, 'hh-register-email')?.attrs?.value === identity.email,
    phone: nodeById(tree, 'hh-register-phone')?.attrs?.value === identity.phone,
    state: selectedOptionValue(tree, 'hh-directory-state') === 'Demo Washington',
    county: selectedOptionValue(tree, 'hh-directory-county') === 'Demo King',
    district: selectedOptionValue(tree, 'hh-directory-district') === 'Demo District',
    school: selectedOptionValue(tree, 'hh-directory-school') === 'Demo Academy',
  };
}
const draftErased = (env) => ['name', 'email', 'phone'].every(field =>
  env.controller.state.journey.registration.identity[field] === '' &&
  nodeById(env.tree, `hh-register-${field}`)?.attrs?.value === '');

function credentialBoundary(env, credentials) {
  const stored = [JSON.stringify(env.controller.state), JSON.stringify(env.runtime.inspect()),
    JSON.stringify(env.runtime.pxc.entries().map(([address, part]) => ({address, value: part.value})))].join('\n');
  return credentials.every(value => !stored.includes(value));
}
