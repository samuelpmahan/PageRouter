const actualFiles = [];
async function walk(dir, prefix = '') {
  for (const item of await readdir(dir, {withFileTypes: true})) {
    const rel = prefix ? `${prefix}/${item.name}` : item.name;
    const full = path.join(dir, item.name);
    if (item.isDirectory()) await walk(full, rel);
    else actualFiles.push(rel);
  }
}

let parsedJsonFiles = 0;
for (const entry of manifest.files) {
  assert.ok(!path.isAbsolute(entry.path) && !entry.path.split('/').includes('..'), `unsafe path: ${entry.path}`);
  const full = path.join(packageRoot, entry.path);
  const stat = await lstat(full);
  assert.ok(stat.isFile() && !stat.isSymbolicLink(), `not a regular file: ${entry.path}`);
  assert.equal(stat.size, entry.bytes, `size mismatch: ${entry.path}`);
  const bytes = await readFile(full);
  assert.equal(sha(bytes), entry.sha256, `hash mismatch: ${entry.path}`);
  if (entry.path.endsWith('.json')) { JSON.parse(bytes.toString('utf8')); parsedJsonFiles++; }
}
