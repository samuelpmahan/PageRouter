function memoryStorage() {
  const values = new Map();
  return {values, getItem(key) { return values.get(key) ?? null; }, setItem(key, value) { values.set(key, String(value)); }, removeItem(key) { values.delete(key); }};
}
function topicFields(tree) {
  const nodes = [];
  const walk = node => { if (!node || typeof node !== 'object') return; nodes.push(node); for (const child of node.children || []) walk(child); };
  walk(tree);
  const form = nodes.find(node => node.tag === 'form' && node.attrs?.['data-submit'] === 'forum-topic');
  if (!form) return null;
  const fields = [];
  const walkForm = node => { if (!node || typeof node !== 'object') return; if (node.attrs?.name) fields.push(node); for (const child of node.children || []) walkForm(child); };
  walkForm(form);
  const title = fields.find(node => node.attrs.name === 'title');
  const category = fields.find(node => node.attrs.name === 'category');
  const content = fields.find(node => node.attrs.name === 'content');
  return {title: title?.attrs.value ?? '', category: category?.children?.find(option => option?.attrs?.selected)?.attrs?.value ?? '',
    content: content?.children?.join('') ?? ''};
}
