function walk(node, visit) {
  if (!node || typeof node !== 'object') return;
  visit(node);
  for (const child of node.children || []) walk(child, visit);
}
function find(node, predicate) {
  let match = null;
  walk(node, candidate => { if (!match && predicate(candidate)) match = candidate; });
  return match;
}
function composerSnapshot() {
  const form = find(emittedTree, node => node.tag === 'form' && node.attrs?.['data-submit'] === 'forum-topic');
  assert.ok(form, 'teacher compose form is present in the emitted page tree');
  const fields = {};
  walk(form, node => {
    if (node.attrs?.name) fields[node.attrs.name] = {
      tag: node.tag,
      valueAttribute: Object.hasOwn(node.attrs, 'value') ? node.attrs.value : null,
      required: node.attrs.required === true,
      options: (node.children || []).filter(child => child?.tag === 'option').map(child => ({value: child.attrs?.value, selected: child.attrs?.selected === true})),
    };
  });
  return {fields, formSubmit: form.attrs['data-submit']};
}
function visibleStatus() {
  return find(emittedTree, node => node.attrs?.role === 'status')?.children?.map(child => typeof child === 'string' ? child : '').join('') || null;
}
